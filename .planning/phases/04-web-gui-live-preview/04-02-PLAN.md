**1. URL → Controls (parse on load):**

Add function `loadFromURL()`:
```typescript
function loadFromURL(): boolean {
  const params = new URLSearchParams(window.location.search);
  if (params.size === 0) return false; // No URL params, use defaults

  // Parse each param, falling back to defaults if missing/invalid
  const rows = parseInt(params.get("rows") ?? "6", 10);
  const cols = parseInt(params.get("cols") ?? "8", 10);
  const w = parseFloat(params.get("w") ?? "297");
  const h = parseFloat(params.get("h") ?? "210");
  const unitParam = params.get("unit") ?? "mm";
  const unit = unitParam === "in" ? "Inches" : "Millimeters";
  const tab = parseInt(params.get("tab") ?? "25", 10) / 100; // 25 → 0.25
  const jitter = parseInt(params.get("jitter") ?? "50", 10) / 100; // 50 → 0.5
  const radius = parseFloat(params.get("radius") ?? "2");
  const kerf = parseFloat(params.get("kerf") ?? "0");
  const seed = params.get("seed") ?? "";

  // Populate inputs
  rowsInput.value = String(rows);
  colsInput.value = String(cols);
  widthInput.value = String(w);
  heightInput.value = String(h);
  unitSelect.value = unit;
  tabSlider.value = String(tab);
  jitterSlider.value = String(jitter);
  radiusSlider.value = String(radius);
  kerfSlider.value = String(kerf);
  seedInput.value = seed || randomSeed(); // Empty seed → generate random

  return true; // Had URL params
}
```

Call `loadFromURL()` in main() BEFORE generating initial puzzle. If it returns false (no params), set seedInput.value = randomSeed() as before. If true, the seed came from URL. Either way, update slider readouts and call generatePuzzle().

**2. Controls → URL (update on every change):**

Add function `updateURL()`:
```typescript
function updateURL(): void {
  const config = buildConfig() as any;
  const params = new URLSearchParams();
  params.set("rows", String(config.rows));
  params.set("cols", String(config.cols));
  params.set("w", String(config.width));
  params.set("h", String(config.height));
  params.set("unit", config.unit === "Inches" ? "in" : "mm");
  params.set("tab", String(Math.round(config.tab.size_pct * 100))); // 0.25 → 25
  params.set("jitter", String(Math.round(config.jitter.amount * 100))); // 0.5 → 50
  params.set("radius", String(config.border.corner_radius));
  params.set("kerf", String(config.kerf_width));
  params.set("seed", config.seed);
  history.replaceState(null, "", "?" + params.toString());
}
```

Call `updateURL()` at the end of `generatePuzzle()` — every parameter change updates the URL via replaceState (no history spam per user decision).

**3. SVG Download:**

Wire the download button (#download):
```typescript
downloadBtn.addEventListener("click", () => {
  const svgContent = svgContainer.innerHTML;
  if (!svgContent) return;
  const config = buildConfig() as any;
  const filename = `puzzle-${config.rows}x${config.cols}-seed-${config.seed}.svg`;
  const blob = new Blob([svgContent], { type: "image/svg+xml" });
  const url = URL.createObjectURL(blob);
  const a = document.createElement("a");
  a.href = url;
  a.download = filename;
  a.click();
  URL.revokeObjectURL(url);
});
```

**4. Copy Link:**

Wire the copy link button (#copy-link):
```typescript
copyLinkBtn.addEventListener("click", async () => {
  try {
    await navigator.clipboard.writeText(window.location.href);
    const original = copyLinkBtn.textContent;
    copyLinkBtn.textContent = "Copied!";
    setTimeout(() => { copyLinkBtn.textContent = original; }, 1500);
  } catch {
    // Fallback: select + copy for non-HTTPS contexts
    const input = document.createElement("input");
    input.value = window.location.href;
    document.body.appendChild(input);
    input.select();
    document.execCommand("copy");
    document.body.removeChild(input);
    const original = copyLinkBtn.textContent;
    copyLinkBtn.textContent = "Copied!";
    setTimeout(() => { copyLinkBtn.textContent = original; }, 1500);
  }
});
```
